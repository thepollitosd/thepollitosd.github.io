        function ctz(bb) { let l = Number(bb & 0xffffffffn); if(l!==0) return 31-Math.clz32(l&-l); let u = Number(bb>>32n); return 63-Math.clz32(u&-u); }

        function getSlidingAttack(sq, occ, dirs) {
            let at = 0n;
            for (let d of dirs) {
                let r = sq >> 3, c = sq & 7;
                while (true) {
                    r += RAY_DIRS[d][0]; c += RAY_DIRS[d][1];
                    if (r < 0 || r > 7 || c < 0 || c > 7) break;
                    let b = bit(r * 8 + c);
                    at |= b;
                    if (occ & b) break;
                }
            }
            return at;
        }

        function initMagics() {
            for (let i = 0; i < 64; i++) {
                let r = i >> 3, c = i & 7;
                // Bishop masks
                let bMask = 0n;
                for (let dr of [-1, 1]) for (let dc of [-1, 1]) {
                    for (let dist = 1; dist < 8; dist++) {
                        let nr = r + dr * dist, nc = c + dc * dist;
                        if (nr > 0 && nr < 7 && nc > 0 && nc < 7) bMask |= bit(nr * 8 + nc);
                    }
                }
                BISHOP_MASKS[i] = bMask;
                // Rook masks
                let rMask = 0n;
                for (let dr of [-1, 0, 1]) for (let dc of [-1, 0, 1]) {
                    if (Math.abs(dr) + Math.abs(dc) !== 1) continue;
                    for (let dist = 1; dist < 8; dist++) {
                        let nr = r + dr * dist, nc = c + dc * dist;
                        if (nr >= 0 && nr < 8 && nc >= 0 && nc < 8) {
                            if (dr !== 0 && nr > 0 && nr < 7) rMask |= bit(nr * 8 + nc);
                            else if (dc !== 0 && nc > 0 && nc < 7) rMask |= bit(nr * 8 + nc);
                        }
                    }
                }
                ROOK_MASKS[i] = rMask;

                // Fill attack tables
                fillAttackTable(i, true);
                fillAttackTable(i, false);
            }
        }

        function fillAttackTable(sq, isBishop) {
            let mask = isBishop ? BISHOP_MASKS[sq] : ROOK_MASKS[sq];
            let bits = countBits(mask);
            let magic = isBishop ? BISHOP_MAGICS[sq] : ROOK_MAGICS[sq];
            let dirs = isBishop ? [1, 3, 5, 7] : [0, 2, 4, 6];
            let table = isBishop ? BISHOP_ATTACKS : ROOK_ATTACKS;

            for (let i = 0; i < (1 << bits); i++) {
                let occ = 0n;
                let tempMask = mask;
                for (let j = 0; j < bits; j++) {
                    let [bitSq, nextMask] = popLSB(tempMask);
                    tempMask = nextMask;
                    if (i & (1 << j)) occ |= bit(bitSq);
                }
                let index = Number(BigInt.asUintN(64, occ * magic.m) >> BigInt(magic.s)) + magic.o;
                table[index] = getSlidingAttack(sq, occ, dirs);
            }
        }

        function getSliding(sq, occ, type) {
            if (type === 2) return getSlidingAttack(sq, occ, DIRS_BISHOP);
            if (type === 3) return getSlidingAttack(sq, occ, DIRS_ROOK);
            return getSlidingAttack(sq, occ, DIRS_QUEEN);
        }

        function setFen(fen) {
            bb.fill(0n); board.fill(-1);
            let parts = fen.split(" ");
            let r=7, c=0;
            let parse = {'P':0,'N':1,'B':2,'R':3,'Q':4,'K':5,'p':6,'n':7,'b':8,'r':9,'q':10,'k':11};
            for(let ch of parts[0]) {
                if(ch==="/") { r--; c=0; }
                else if(ch>='1' && ch<='8') c += parseInt(ch);
                else {
                    let p = parse[ch];
                    bb[p] |= bit(r*8+c); board[r*8+c] = p; c++;
                }
            }
            turn = parts[1]==='b'?1:0;
            castle = 0;
            if(parts[2].includes('K')) castle |= 1;
            if(parts[2].includes('Q')) castle |= 2;
            if(parts[2].includes('k')) castle |= 4;
            if(parts[2].includes('q')) castle |= 8;
            ep = parts[3]==='-' ? -1 : (parts[3].charCodeAt(0)-97) + (parseInt(parts[3][1])-1)*8;
            calcHash();
            repetitionStack = [hash];
        }

        function wOcc() { return bb[0]|bb[1]|bb[2]|bb[3]|bb[4]|bb[5]; }
        function bOcc() { return bb[6]|bb[7]|bb[8]|bb[9]|bb[10]|bb[11]; }

        function calcHash() {
            hash = 0n;
            for(let i=0; i<64; i++) if(board[i]!==-1) hash ^= ZOBRIST[board[i]*64+i];
            if(turn===1) hash ^= ZOB_BLACK;
            hash ^= ZOB_CASTLE[castle];
            if(ep!==-1) hash ^= ZOB_EP[ep&7];
        }

        function isAttacked(sq, byBlack, occ = null) {
            if (occ === null) occ = wOcc() | bOcc();
            let o = byBlack ? 6 : 0;
            if (PAWN_ATTACKS[byBlack ? 0 : 1][sq] & bb[o]) return true;
            if (KNIGHT_ATTACKS[sq] & bb[o+1]) return true;
            if (KING_ATTACKS[sq] & bb[o+5]) return true;
            if (getSliding(sq, occ, 2) & (bb[o+2]|bb[o+4])) return true;
            if (getSliding(sq, occ, 3) & (bb[o+3]|bb[o+4])) return true;
            return false;
        }

        function genMoves(capturesOnly = false, quietOnly = false) {
            let moves = [];
            let o = turn?6:0;
            let occ = wOcc() | bOcc();
            let enemyO = turn ? wOcc() : bOcc();
            let allyO = turn ? bOcc() : wOcc();
            let empty = ~occ;
            
            // Pawns
            let p = bb[o];
            while(p) {
                let [sq, nP] = popLSB(p); p = nP;
                let r = sq>>3;
                let pushSq = sq + (turn?-8:8);
                if (!capturesOnly && (empty & bit(pushSq))) {
                    if ((turn && r===1) || (!turn && r===6)) { // promotion
                        [o+1,o+2,o+3,o+4].forEach(prom => moves.push({f:sq, t:pushSq, c:-1, pce:o, prom, flag:0}));
                    } else {
                        moves.push({f:sq, t:pushSq, c:-1, pce:o, prom:-1, flag:0});
                        let dblSq = sq + (turn?-16:16);
                        if (((turn && r===6) || (!turn && r===1)) && (empty & bit(dblSq))) moves.push({f:sq, t:dblSq, c:-1, pce:o, prom:-1, flag:4});
                    }
                }
                let att = PAWN_ATTACKS[turn][sq] & enemyO;
                while(att) {
                    let [tsq, nAtt] = popLSB(att); att = nAtt;
                    if (!quietOnly) {
                        if ((turn && r===1) || (!turn && r===6)) {
                            [o+1,o+2,o+3,o+4].forEach(prom => moves.push({f:sq, t:tsq, c:board[tsq], pce:o, prom, flag:0}));
                        } else {
                            moves.push({f:sq, t:tsq, c:board[tsq], pce:o, prom:-1, flag:0});
                        }
                    }
                }
                if (ep !== -1 && !quietOnly) {
                    if (PAWN_ATTACKS[turn][sq] & bit(ep)) {
                        moves.push({f:sq, t:ep, c:board[ep + (turn?8:-8)], pce:o, prom:-1, flag:1});
                    }
                }
            }

            // Knights & King
            let pts = [ [o+1, KNIGHT_ATTACKS], [o+5, KING_ATTACKS] ];
            for(let [pce, attMap] of pts) {
                let p = bb[pce];
                while(p) {
                    let [sq, nP] = popLSB(p); p = nP;
                    let m = attMap[sq] & ~allyO;
                    if(capturesOnly) m &= enemyO;
                    while(m) {
                        let [tsq, nM] = popLSB(m); m = nM;
                        if (quietOnly && board[tsq] !== -1) continue;
                        if (capturesOnly && board[tsq] === -1) continue;
                        moves.push({f:sq, t:tsq, c:board[tsq], pce:pce, prom:-1, flag:0});
                    }
                }
            }

            // Sliding
            for(let t=2; t<=4; t++) {
                let p = bb[o+t];
                while(p) {
                    let [sq, nP] = popLSB(p); p = nP;
                    let m = getSliding(sq, occ, t) & ~allyO;
                    if(capturesOnly) m &= enemyO;
                    while(m) {
                        let [tsq, nM] = popLSB(m); m = nM;
                        if (quietOnly && board[tsq] !== -1) continue;
                        if (capturesOnly && board[tsq] === -1) continue;
                        moves.push({f:sq, t:tsq, c:board[tsq], pce:o+t, prom:-1, flag:0});
                    }
                }
            }

            // Castling
            if (!capturesOnly) {
                if(turn===0) {
                    if ((castle&1) && !(occ&0x60n) && !isAttacked(4,1) && !isAttacked(5,1) && !isAttacked(6,1)) moves.push({f:4, t:6, c:-1, pce:5, prom:-1, flag:2});
                    if ((castle&2) && !(occ&0xEn) && !isAttacked(4,1) && !isAttacked(3,1) && !isAttacked(2,1)) moves.push({f:4, t:2, c:-1, pce:5, prom:-1, flag:2});
                } else {
                    if ((castle&4) && !(occ&(0x60n<<56n)) && !isAttacked(60,0) && !isAttacked(61,0) && !isAttacked(62,0)) moves.push({f:60, t:62, c:-1, pce:11, prom:-1, flag:2});
                    if ((castle&8) && !(occ&(0xEn<<56n)) && !isAttacked(60,0) && !isAttacked(59,0) && !isAttacked(58,0)) moves.push({f:60, t:58, c:-1, pce:11, prom:-1, flag:2});
                }
            }

            return moves;
        }


        // Legal move filter: wraps genMoves and removes moves that leave king in check
        function genLegalMoves(capturesOnly = false, quietOnly = false) {
            let pseudo = genMoves(capturesOnly, quietOnly);
            let legal = [];
            for (let m of pseudo) {
                makeMove(m);
                if (!isAttacked(ctz(bb[turn?5:11]), turn, wOcc() | bOcc())) legal.push(m);
                unmakeMove(m);
            }
            return legal;
        }

        // TIER 5 / TIER 1 Move Making + Zobrist hashing incrementally
        function makeMove(m) {
            pastStates.push({ep, castle, hash});
            hash ^= ZOB_CASTLE[castle];
            if(ep!==-1) hash ^= ZOB_EP[ep&7];

            let {f, t, pce, c, prom, flag} = m;
            bb[pce] ^= bit(f); board[f] = -1; hash ^= ZOBRIST[pce*64 + f];

            if (c !== -1) {
                if (flag===1) { // EP capture
                    let cs = t + (turn?8:-8);
                    bb[c] ^= bit(cs); board[cs] = -1; hash ^= ZOBRIST[c*64 + cs];
                } else {
                    bb[c] ^= bit(t); hash ^= ZOBRIST[c*64 + t];
                }
            }
            let np = prom!==-1 ? prom : pce;
            bb[np] |= bit(t); board[t] = np; hash ^= ZOBRIST[np*64 + t];

            if (flag===2) { // Castling rook moves
                if(t===6){ bb[3]^=bit(7)|bit(5); board[7]=-1; board[5]=3; hash^=ZOBRIST[3*64+7]^ZOBRIST[3*64+5]; }
                else if(t===2){ bb[3]^=bit(0)|bit(3); board[0]=-1; board[3]=3; hash^=ZOBRIST[3*64+0]^ZOBRIST[3*64+3]; }

        function evaluate() {
            let mgWhite = 0, mgBlack = 0;
            let egWhite = 0, egBlack = 0;
            let gamePhase = 0;

            for (let p = 0; p < 12; p++) {
                let mask = bb[p];
                let isWhite = p < 6;
                while (mask) {
                    let [sq, nM] = popLSB(mask); mask = nM;
                    if (isWhite) {
                        mgWhite += mg_table[p * 64 + sq];
                        egWhite += eg_table[p * 64 + sq];
                    } else {
                        mgBlack += mg_table[p * 64 + sq];
                        egBlack += eg_table[p * 64 + sq];
                    }
                    gamePhase += PESTO_GAMEPHASE_INC[p % 6];
                }
            }

            let mgScore = mgWhite - mgBlack;
            let egScore = egWhite - egBlack;
            let mgPhase = gamePhase;
            if (mgPhase > 24) mgPhase = 24;
            let egPhase = 24 - mgPhase;
            
            let score = (mgScore * mgPhase + egScore * egPhase) / 24;
            
            let finalScore = turn === 0 ? score : -score;
            // tempo
            finalScore += 10;
            return Math.floor(finalScore);
        }

        // Static Exchange Evaluation: returns the material gain/loss after all captures on a square.
        // Uses the standard gain[] minimax approach so each side can stop capturing if losing.
        function see(move) {
            const PIECE_VAL = [100, 320, 330, 500, 900, 20000, 100, 320, 330, 500, 900, 20000];
            let to = move.t;
            let from = move.f;
            let capturedVal = move.c !== -1 ? PIECE_VAL[move.c] : 0;
            let movingPieceVal = PIECE_VAL[move.pce];

            // gain[0] = value captured by moving piece
            let gain = [capturedVal];
            let d = 0;
            let occ = wOcc() | bOcc();
            occ ^= bit(from); // remove moving piece

            // Which side to move next (opponent of moving piece)
            let nextSide = move.pce < 6 ? 1 : 0; // 0=white, 1=black
            let lastPieceVal = movingPieceVal;

            while (true) {
                // Find least valuable attacker for nextSide on 'to'
                let found = false;
                let bestVal = Infinity, bestSq = -1, bestP = -1;
                let start = nextSide === 0 ? 0 : 6;
                let end = nextSide === 0 ? 6 : 12;
                for (let p = start; p < end; p++) {
                    let mask = bb[p] & occ;
                    while (mask) {
                        let [sq, nM] = popLSB(mask); mask = nM;
                        let attacks = 0n;
                        let pi = p % 6;
                        if (pi === 0) attacks = PAWN_ATTACKS[p < 6 ? 0 : 1][sq];
                        else if (pi === 1) attacks = KNIGHT_ATTACKS[sq];
                        else if (pi === 2) attacks = getSliding(sq, occ, 2);
                        else if (pi === 3) attacks = getSliding(sq, occ, 3);
                        else if (pi === 4) attacks = getSliding(sq, occ, 0);
                        else if (pi === 5) attacks = KING_ATTACKS[sq];
                        if (attacks & bit(to)) {
                            if (PIECE_VAL[p] < bestVal) {
                                bestVal = PIECE_VAL[p]; bestSq = sq; bestP = p; found = true;
                            }
                        }
                    }
                }
                if (!found) break;

                d++;
                gain.push(lastPieceVal - gain[d - 1]);
                lastPieceVal = bestVal;
                occ ^= bit(bestSq); // remove this attacker
                nextSide ^= 1;
            }

            // Minimax backward: each side picks the better of capturing or not
            for (let i = gain.length - 1; i > 0; i--) {
                gain[i - 1] = Math.max(-gain[i], gain[i - 1]);
            }
            return gain[0];
        }

        function getTT(key) {
            if (!TT_VIEW) return null;
            let idx = Number(key & BigInt(TT_MASK)) * 4;
            let kLow = Atomics.load(TT_VIEW, idx);
            let kHigh = Atomics.load(TT_VIEW, idx + 1);
            if (kLow !== Number(key & 0xffffffffn) || kHigh !== Number(key >> 32n)) return null;
            let data1 = Atomics.load(TT_VIEW, idx + 2); // score
            let data2 = Atomics.load(TT_VIEW, idx + 3); // depth | flag | move
            let f = (data2 >> 10) & 0x3F;
            let t = (data2 >> 4) & 0x3F;
            let promRaw = data2 & 0x0F;
            let move = (f === 0x3F && t === 0x3F) ? null : {
                f, t,
                prom: promRaw === 0x0F ? -1 : promRaw
            };
            return {
                key: key,
                score: data1,
                depth: (data2 >> 24) & 0xFF,
                flag: (data2 >> 16) & 0xFF,
                move
            };
        }

        function saveTT(key, score, depth, flag, move) {
            if (!TT_VIEW) return;

            // Replacement strategy: depth-preferred
            let oldData2 = Atomics.load(TT_VIEW, idx + 3);
            let oldDepth = (oldData2 >> 24) & 0xFF;
            if (oldDepth > depth) return;

            let moveData = 0;
            if (move) {
                let prom = move.prom === -1 ? 0x0F : move.prom;
                moveData = (move.f << 10) | (move.t << 4) | prom;

                moveData = (0x3F << 10) | (0x3F << 4) | 0x0F;

            Atomics.store(TT_VIEW, idx, Number(key & 0xffffffffn));
            Atomics.store(TT_VIEW, idx + 1, Number(key >> 32n));
            Atomics.store(TT_VIEW, idx + 2, score);
            Atomics.store(TT_VIEW, idx + 3, data2);
        }

        function search(depth, rootMoves, alpha, beta, startDepth = 1) {
            nodes=0; timeUp=false; startTime = Date.now();
            let bMove = null, bScore = -Infinity;
            resetHeuristics();
            // Iterative Deepening with Aspiration Windows
            for(let d = startDepth; d <= depth; d++) {
                if(timeUp && d > minSearchDepth) break;
                if (rootMoves && bMove) {
                    rootMoves.sort((a, b) => (moveEquals(a, bMove) ? 1 : 0) - (moveEquals(b, bMove) ? 1 : 0));
                    rootMoves.reverse();
                }
                // Wider aspiration windows reduce costly re-searches while still catching big swings
                let a = d>3 ? bScore - 150 : -50000;
                let b = d>3 ? bScore + 150 : 50000;
                
                while(true) {
                    let {score, move} = alphaBeta(d, 0, rootMoves||null, a, b);
                    if(timeUp && d > minSearchDepth) break;
                    if(score <= a) { a = Math.max(a - 500, -50000); }
                    else if(score >= b) { b = Math.min(b + 500, 50000); }
                    else {
                        bScore = score;
                        if(move) bMove = move;
                        // Always report PV info, including id and current depth to differentiate lines
                        postMessage({status:'info', depth:d, maxDepth:depth, score: bScore, move: fmt(bMove), nodes, id: self.workerId, token: self.searchToken});
                        break;
                    }
                }
                if (bScore > 40000) break; // Early exit on forced mate
                if (Date.now() - startTime > softLimitMs && d >= minSearchDepth) break; // Soft bound finished depth
            }
            
            // TIER 5: Blunder Check & Focus
            if (bMove && bScore < -300) {
                postMessage({status:'threat', msg: 'Tactical threat detected!', id: self.workerId, token: self.searchToken});
            }

            return {move: bMove, score: bScore, nodes};
        }

                table[v] = [];
                for (let a = 0; a < 6; a++) {
                    table[v][a] = vals[v] * 10 - vals[a];
                }
            }
            return table;
        })();

        function scoreMove(m, ttMove, ply, lastMove) {
            if (ttMove && moveEquals(m, ttMove)) return 2000000; // TT move always first
            if (m.prom === 4 || m.prom === 10) return 900000;   // Queen promotion
            if (m.c !== -1) {
                // Captures: MVV-LVA. Always score above quiet moves.
                return 100000 + MVV_LVA[m.c % 6][m.pce % 6];
            }
            // Quiet moves
            if (m.flag === 2) return 90000; // Castling
            if (m.pce === 5 && m.flag !== 2) return -10000; // King walk penalty
            if (killerMoves[ply] && killerMoves[ply][0] && moveEquals(killerMoves[ply][0], m)) return 80000;
            if (killerMoves[ply] && killerMoves[ply][1] && moveEquals(killerMoves[ply][1], m)) return 70000;
            if (lastMove && counterMoves[lastMove.pce] && counterMoves[lastMove.pce][lastMove.t] && moveEquals(counterMoves[lastMove.pce][lastMove.t], m)) return 60000;
            return history[turn][m.f][m.t] || 0;
        }

        function alphaBeta(depth, ply, rootMoves, alpha, beta, lastMove = null) {
            // TIER 5: Mate Distance Pruning
            let mateVal = 40000 - ply;
            if (alpha < -mateVal) alpha = -mateVal;
            if (beta > mateVal - 1) beta = mateVal - 1;
            if (alpha >= beta) return {score: alpha, move: null};

            if((nodes++ & 1023)===0) {
                let now = Date.now();
                if(now - startTime > limitMs && depth > minSearchDepth) timeUp = true;
                else if(now - lastBrainstorm >= 100) {
                    lastBrainstorm = now;
                    postMessage({status: 'brainstorm', fen: exportFEN(), id: self.workerId, token: self.searchToken});
                }
            }
            if(timeUp && depth > minSearchDepth) return {score: alpha, move: null};
            if (ply > 0 && isRepetition()) return {score: 0, move: null};

            let occ = wOcc() | bOcc();
            let inCheck = isAttacked(ctz(bb[turn?11:5]), turn^1, occ);
            currentMaterialBalance = 0;
            for (let p = 0; p < 6; p++) currentMaterialBalance += (countBits(bb[p]) - countBits(bb[p+6])) * V[p];

            // Check Extension: Search deeper when in check to avoid blunders
            if (inCheck) depth += 1;

            // Zobrist TT
            let tte = getTT(hash);
            // FIX: Never cutoff at the root (ply > 0) to avoid playing corrupted moves from SharedArrayBuffer data races.
            if(tte && tte.depth >= depth && ply > 0) {
                if(tte.flag === 0) return {score: tte.score, move: tte.move};
                if(tte.flag === 1 && tte.score <= alpha) return {score: tte.score, move: tte.move};
                if(tte.flag === 2 && tte.score >= beta) return {score: tte.score, move: tte.move};
            }

            // IID removed: wasted more time than it saved

            if(depth <= 0) return {score: quiesce(alpha, beta), move: null};

            // Reverse Futility Pruning: only at depth 1-2, with a generous margin to avoid blunders
            if (depth <= 2 && !inCheck && Math.abs(alpha) < 40000) {
                let staticEval = evaluateFast();
                if (staticEval - depth * 250 >= beta) return {score: staticEval, move: null};
            }

            // Razoring: removed — was causing blunders by skipping quiet moves too early

            // NMP: require depth>=4, skip in PV nodes and when few pieces remain (zugzwang risk)
            let isPV = (beta - alpha > 1);
            if (depth >= 4 && !inCheck && ply > 0 && !isPV) {
                // Rough zugzwang guard: don't null-move if side has only king+pawns
                let sideOcc = turn ? bOcc() : wOcc();
                let sideHasPieces = (sideOcc & ~(bb[turn?12:6] | bb[turn?6:0])) !== 0n;
                if (sideHasPieces) {
                    let oldTurn = turn, oldEp = ep, oldHash = hash;
                    turn ^= 1; ep = -1; calcHash();
                    let s = -alphaBeta(depth - 3 - (depth>6 ? 1 : 0), ply+1, null, -beta, -beta+1, null).score;
                    turn = oldTurn; ep = oldEp; hash = oldHash;
                    if (s >= beta) return {score: beta, move: null};
                }
            }

            let ttMove = tte && tte.move ? tte.move : null;
            // Always generate captures and quiet moves separately so captures are always searched first
            let captureMoves, quietMoves;
            if (rootMoves) {
                // Root: moves already fully generated and sorted externally
                captureMoves = rootMoves.filter(m => m.c !== -1 || m.prom !== -1);
                quietMoves   = rootMoves.filter(m => m.c === -1 && m.prom === -1);
                // Re-score with TT move priority
                [...captureMoves, ...quietMoves].forEach(m => { m.s = scoreMove(m, ttMove, ply, lastMove); });
                captureMoves.sort((a,b) => b.s - a.s);
                quietMoves.sort((a,b) => b.s - a.s);
            } else {
                captureMoves = genMoves(true, false);
                captureMoves.forEach(m => { m.s = scoreMove(m, ttMove, ply, lastMove); });
                captureMoves.sort((a,b) => b.s - a.s);
                quietMoves = genMoves(false, true);
                quietMoves.forEach(m => { m.s = scoreMove(m, ttMove, ply, lastMove); });
                quietMoves.sort((a,b) => b.s - a.s);
            }

            if (captureMoves.length + quietMoves.length === 0) return {score: inCheck ? -40000+ply : 0, move: null};

            let bestMove = null;
            let origAlpha = alpha;
            let movesMade = 0;
            // Always do captures first, then quiet — two stages regardless of root or not
            let moveStages = [captureMoves, quietMoves];

            for (let stage = 0; stage < moveStages.length; stage++) {
                let moves = moveStages[stage];
                for (let m of moves) {
                    makeMove(m);
                    if(isAttacked(ctz(bb[turn?5:11]), turn, (wOcc() | bOcc()))) { unmakeMove(m); continue; } // illegal
                    movesMade++;
                    let score;
                    let isQuiet = (m.c === -1 && m.prom === -1);
                    let isTt = ttMove && moveEquals(m, ttMove);
                    let isKiller = killerMoves[ply] && ((killerMoves[ply][0] && moveEquals(killerMoves[ply][0], m)) || (killerMoves[ply][1] && moveEquals(killerMoves[ply][1], m)));

                    // TIER 1: PVS
                    if (movesMade === 1) {
                        score = -alphaBeta(depth-1, ply+1, null, -beta, -alpha, m).score;
                    } else {
                        let isCapture = (m.c !== -1 || m.prom !== -1);

                        // Never LMP or LMR captures — always search them fully
                        if (!isCapture) {
                            // LMP: skip late quiet moves at low depth
                            let lmpThreshold = 5 + 3 * depth * depth;
                            if(depth <= 3 && movesMade > lmpThreshold && !inCheck) {
                                unmakeMove(m); continue;
                            }

                            // LMR: reduce late quiet moves
                            let doLMR = (depth >= 3 && movesMade >= 6 && !inCheck && !isTt && !isKiller);
                            if (doLMR) {
                                let reduction = Math.max(1, Math.floor(Math.log(depth) * Math.log(movesMade) / 2.5));
                                score = -alphaBeta(depth - 1 - reduction, ply+1, null, -alpha-1, -alpha, m).score;
                                if (score > alpha) score = -alphaBeta(depth-1, ply+1, null, -beta, -alpha, m).score;
                            } else {
                                score = -alphaBeta(depth-1, ply+1, null, -alpha-1, -alpha, m).score;
                                if(score > alpha && score < beta) score = -alphaBeta(depth-1, ply+1, null, -beta, -alpha, m).score;
                            }
                        } else {
                            // Captures: PVS but no reduction
                            score = -alphaBeta(depth-1, ply+1, null, -alpha-1, -alpha, m).score;
                            if(score > alpha && score < beta) score = -alphaBeta(depth-1, ply+1, null, -beta, -alpha, m).score;
                        }
                    }
                    unmakeMove(m);
                    if(timeUp) return {score: alpha, move: bestMove};

                    if(score > alpha) {
                        alpha = score; bestMove = m;
                        if(score >= beta) {
                            if (m.c === -1 && m.prom === -1) {
                                let bonus = depth * depth;
                                history[turn][m.f][m.t] += bonus - (history[turn][m.f][m.t] * Math.abs(bonus) / 324);
                                if (!killerMoves[ply][0] || (killerMoves[ply][0].f !== m.f || killerMoves[ply][0].t !== m.t)) {
                                    killerMoves[ply][1] = killerMoves[ply][0];
                                    killerMoves[ply][0] = m;
                                }
                            }
                            if (lastMove) counterMoves[lastMove.pce][lastMove.t] = m;
                            break;
                        }
                    }
                }
                if (alpha >= beta) break;
                if (timeUp) return {score: alpha, move: bestMove};
            }

            if(movesMade===0) return {score: inCheck ? -40000+ply : 0, move: null};

            saveTT(hash, alpha, depth, alpha<=origAlpha ? 1 : (alpha>=beta ? 2 : 0), bestMove);

            return {score: alpha, move: bestMove};
        }

        // TIER 2: Quiescence — searches all captures and check evasions until position is quiet
        function quiesce(alpha, beta) {
            if((nodes++ & 1023)===0 && Date.now() - startTime > limitMs) timeUp = true;
            if(timeUp) return alpha;

            let stand_pat = evaluate();
            if(stand_pat >= beta) return beta;
            if(stand_pat > alpha) alpha = stand_pat;

            let inCheck = isAttacked(ctz(bb[turn?5:11]), turn, (wOcc() | bOcc()));
            let moves;
            if (inCheck) {
                // In check: must consider all moves to find evasions
                moves = genMoves(false);
                moves.forEach(m => { m.s = scoreMove(m, null, 0, null); });
                moves.sort((a, b) => b.s - a.s);
            } else {
                // Not in check: only captures (and promotions)
                moves = genMoves(true);
                // Sort by MVV-LVA: most valuable victim, least valuable attacker
                moves.sort((a, b) =>
                    ((V[b.c % 6] || 0) * 10 - (V[b.pce % 6] || 0)) -
                    ((V[a.c % 6] || 0) * 10 - (V[a.pce % 6] || 0))
                );
            }

            for (let m of moves) {
                // Delta pruning: skip captures that can't raise alpha even if recaptured optimally
                // Only apply when NOT in check and margin is generous
                if (!inCheck && m.c !== -1 && m.prom === -1) {
                    if (stand_pat + (V[m.c % 6] || 0) + 350 < alpha) continue;
                }

                makeMove(m);
                if(isAttacked(ctz(bb[turn?5:11]), turn, (wOcc() | bOcc()))) { unmakeMove(m); continue; }
                let s = -quiesce(-beta, -alpha);
                unmakeMove(m);
                if(timeUp) return alpha;
                if(s >= beta) return beta;
                if(s > alpha) alpha = s;
            }
            return alpha;
        }

        initPrecomputes();

        self.onmessage = function(e) {
            let msg = e.data;
            if (msg.cmd === 'init') {
                TT_BUFFER = msg.ttBuffer;
                TT_VIEW = new Int32Array(TT_BUFFER);
            } else if (msg.cmd === 'stop') {
                if (msg.token === undefined || msg.token === self.searchToken) {
                    timeUp = true;
                    limitMs = 0;
                    softLimitMs = 0;
                }
            } else if (msg.cmd === 'search') {
                setFen(msg.fen);
                self.searchToken = msg.token;
                let t = Number(msg.time);
                if (!Number.isFinite(t) || t <= 0) t = 1.5;
                softLimitMs = t * 1000;
                limitMs = Math.max(softLimitMs * 1.5, softLimitMs + 1000);
                minSearchDepth = Number(msg.minDepth) || 1;
                useNN = msg.useNN !== false;
                self.workerId = msg.id; 
                let rootMoves = genLegalMoves(false, false);
                rootMoves.forEach(m => { m.s = scoreMove(m, null, 0, null); });
                rootMoves.sort((a, b) => b.s - a.s);
                // Lazy SMP: stagger starting depth so workers don't all hit the same depth simultaneously
                let startDepth = 1 + (self.workerId % 3);
                let res = search(64, rootMoves, -50000, 50000, startDepth);
                postMessage({ status: 'done', move: fmt(res.move), score: res.score, nodes: res.nodes, id: msg.id, token: self.searchToken });
            }
        };
